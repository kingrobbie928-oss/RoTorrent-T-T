target_link_libraries(rotorrent_base
    PRIVATE
        OpenSSL::Crypto
        OpenSSL::SSL
        # find whatever zlib target that exists while being compatible with both
        # FindZLIB (from cmake) and zlib own scripts
        $<IF:$<TARGET_EXISTS:ZLIB::ZLIB>,ZLIB::ZLIB,ZLIB::ZLIBSTATIC>
    PUBLIC
        LibtorrentRasterbar::torrent-rasterbar
        Qt::Core
        Qt::CorePrivate
        Qt::Network
        Qt::Sql
        rotorrent_common_cfg
)

if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    include(CheckSymbolExists)
    check_symbol_exists(getrandom "sys/random.h" HAVE_GETRANDOM)

    if (HAVE_GETRANDOM)
        target_compile_definitions(rotorrent_base PUBLIC QBT_USES_GETRANDOM)
    endif()
elseif (CMAKE_SYSTEM_NAME STREQUAL "Windows")
    target_sources(rotorrent_base PRIVATE
        utils/reg.h
        utils/reg.cpp
    )
    target_link_libraries(rotorrent_base PRIVATE iphlpapi powrprof)
endif()
